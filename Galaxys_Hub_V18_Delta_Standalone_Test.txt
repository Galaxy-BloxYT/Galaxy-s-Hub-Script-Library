local function themeColor(role)
    return THEME_PALETTES[activeTheme][role]

local function applyTheme(name)
    if not THEME_PALETTES[name] then
        return false
    end

    activeTheme = name
    for instance, properties in pairs(themedProperties) do
        if instance.Parent then
            for property, role in pairs(properties) do
                pcall(function()
                    instance[property] = themeColor(role)
                end)
            end
        else
            themedProperties[instance] = nil
        end
    end
    return true
end

	local object = Instance.new(className)

local function makeDraggable(handle, target)
	local dragging = false
	local dragInput
	local dragStart
	local startPosition

	handle.Active = true

	handle.InputBegan:Connect(function(input)
		if not isPrimaryInput(input) then
			return
		end

		dragging = true
		dragStart = input.Position
		startPosition = target.Position
		dragInput = input

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)

    local dragConnection
    dragConnection = UserInputService.InputChanged:Connect(function(input)
        if not target.Parent then
            dragConnection:Disconnect()
            return
        end
        if not dragging or input ~= dragInput or not dragStart or not startPosition then
			return
		end

		local delta = input.Position - dragStart
		target.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)
	end)
end

	function Window:SelectTab(tab)
		local target = tab

		if type(tab) == "string" then
			target = self._tabs[tab]
		end

		if type(target) ~= "table" or not target.Frame then
			return false
		end

		for _, tabObject in pairs(self._tabs) do
			if type(tabObject) == "table" and tabObject.Frame then
				tabObject.Frame.Visible = false
				tabObject.Button.BackgroundColor3 = themeColor("Raised")
			end
		end

		target.Frame.Visible = true
        target.Button.BackgroundColor3 = themeColor("Accent")
        for _, tabObject in pairs(self._tabs) do
            if tabObject.Button then
                themedProperties[tabObject.Button] = themedProperties[tabObject.Button] or {}
                themedProperties[tabObject.Button].BackgroundColor3 =
                    tabObject == target and "Accent" or "Raised"
            end
        end
        self._currentTab = target
		return true
	end

				Fill.Size = UDim2.new(normalized, 0, 1, 0)

					safeCallback(sliderCallback, value)

				if BarBackground.AbsoluteSize.X <= 0 then
					return

				setFromRatio(ratio, fireCallback)
			end

				if isPrimaryInput(input) then
					draggingSlider = false
				end
			end)

				setFromRatio(ratio, fireCallback)
			end

				return value
			end

			return controller
		end

					elseif #selected == 1 then return optionLabel(selected[1]) .. " ▾"
					else return tostring(#selected) .. " selected ▾" end
				else
					return optionLabel(selected) .. " ▾"
				end
			end

	            local function sameOption(left, right)
                return left == right or optionLabel(left) == optionLabel(right)
            end

            local function isSelected(option)
				if multiple then
					for _, v in ipairs(selected) do if sameOption(v, option) then return true end end
					return false
				else
					return sameOption(selected, option)
				end
			end

			local function buildOptions()
				for _, child in ipairs(OptionsScroll:GetChildren()) do
					if child:IsA("TextButton") then child:Destroy() end
				end

				for _, option in ipairs(options) do
					local strOpt = optionLabel(option)
					if filterText == "" or string.find(string.lower(strOpt), string.lower(filterText)) then
						local active = isSelected(option)

						local OptButton = create("TextButton", {
							Size = UDim2.new(1, 0, 0, 28),
							BackgroundColor3 = active and Color3.fromRGB(45, 45, 60) or Color3.fromRGB(25, 25, 31),
							BorderSizePixel = 0,
							Text = "  " .. (multiple and (active and "[✓] " or "[ ] ") or "") .. strOpt,
							TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(180, 180, 190),
							TextSize = 12,
							Font = Enum.Font.Gotham,
							TextXAlignment = Enum.TextXAlignment.Left,
							AutoButtonColor = false,
						}, OptionsScroll)

						OptButton.MouseButton1Click:Connect(function()
							if multiple then
								local foundIndex = nil
								for i, v in ipairs(selected) do
									if sameOption(v, option) then foundIndex = i; break; end
								end
								if foundIndex then table.remove(selected, foundIndex)
								else table.insert(selected, option) end

								ValueIndicator.Text = getDisplayText()
								buildOptions()
								safeCallback(callback, multiple and table.clone(selected) or selected)
							else
								selected = option
								ValueIndicator.Text = getDisplayText()
								opened = false
								
								TweenService:Create(DropdownHolder, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
									Size = UDim2.new(1, 0, 0, 38)
								}):Play()
								TweenService:Create(ListContainer, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
									Size = UDim2.new(1, 0, 0, 0)
								}):Play()
								task.delay(0.2, function() ListContainer.Visible = false end)

								buildOptions()
								safeCallback(callback, multiple and table.clone(selected) or selected)
							end
						end)
					end
				end
			end

				SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
				filterText = SearchBox.Text
				buildOptions()
			end)

			DropdownButton.MouseButton1Click:Connect(function()
				opened = not opened
				
				local visibleItemsCount = math.min(#options, 4)
				local targetListHeight = (visibleItemsCount * 30) + 44
				local targetHolderHeight = 38 + targetListHeight + 6

				if opened then
					ListContainer.Visible = true
					filterText = ""
					SearchBox.Text = ""
					buildOptions()

					TweenService:Create(DropdownHolder, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
						Size = UDim2.new(1, 0, 0, targetHolderHeight)
					}):Play()
					TweenService:Create(ListContainer, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
						Size = UDim2.new(1, 0, 0, targetListHeight)
					}):Play()
				else
					TweenService:Create(DropdownHolder, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
						Size = UDim2.new(1, 0, 0, 38)
					}):Play()
					TweenService:Create(ListContainer, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
						Size = UDim2.new(1, 0, 0, 0)
					}):Play()
					task.delay(0.2, function()
						if not opened then ListContainer.Visible = false end
					end)
				end
			end)

			local controller = {}
			function controller:Set(value)
				if multiple then
					if type(value) == "table" then
						selected = {}
						for _, v in ipairs(value) do table.insert(selected, v) end
						ValueIndicator.Text = getDisplayText()
						buildOptions()
						safeCallback(callback, multiple and table.clone(selected) or selected)
					end
				else
					if table.find(options, value) or type(value) == "string" then
						selected = value
						ValueIndicator.Text = getDisplayText()
						buildOptions()
						safeCallback(callback, multiple and table.clone(selected) or selected)
					end
				end
			end
			function controller:Get()
				return selected
			end

			controller.Instance = DropdownHolder
			return controller
		end

					return
				end

				saturation = math.clamp(
					(position.X - SaturationValue.AbsolutePosition.X) / SaturationValue.AbsoluteSize.X,
					0,
					1
				)

				value = 1 - math.clamp(
					(position.Y - SaturationValue.AbsolutePosition.Y) / SaturationValue.AbsoluteSize.Y,
					0,
					1
				)

				updateColor(true)
			end

			local function updateHue(position)
				if HueBar.AbsoluteSize.Y <= 0 then
					return
				end

				hue = math.clamp(
					(position.Y - HueBar.AbsolutePosition.Y) / HueBar.AbsoluteSize.Y,
					0,
					1
				)

				updateColor(true)
			end

			Preview.MouseButton1Click:Connect(function()
				pickerOpen = not pickerOpen
				Picker.Visible = pickerOpen
				Holder.Size = pickerOpen
					and UDim2.new(1, 0, 0, 188)
					or UDim2.new(1, 0, 0, 40)
			end)

			SaturationValue.InputBegan:Connect(function(input)
				if isPrimaryInput(input) then
					draggingSV = true
					updateSV(input.Position)
				end
			end)

			HueBar.InputBegan:Connect(function(input)
				if isPrimaryInput(input) then
					draggingHue = true
					updateHue(input.Position)
				end
			end)

			UserInputService.InputChanged:Connect(function(input)
				if input.UserInputType ~= Enum.UserInputType.MouseMovement
					and input.UserInputType ~= Enum.UserInputType.Touch then
					return
				end

				if draggingSV then
					updateSV(input.Position)
				elseif draggingHue then
					updateHue(input.Position)
				end
			end)

			UserInputService.InputEnded:Connect(function(input)
				if isPrimaryInput(input) then
					draggingSV = false
					draggingHue = false
				end
			end)

			updateColor(false)

			local controller = {}

			function controller:Set(color, fireCallback)
				if typeof(color) ~= "Color3" then
					return
				end

				hue, saturation, value = color:ToHSV()
				updateColor(fireCallback)
			end

			function controller:Get()
				return selectedColor
			end

			controller.Instance = Holder
			return controller
		end

    		if not self._currentTab then
			self:SelectTab(TabObj)
		end

		return TabObj
	end

	return Window
end


-- Static GUI key system.
-- Returns true only after the correct key is entered.
function Library:InitKeySystem(config)
	config = config or {}

	if config.UseKeySystem == false then
		return true
	end

	local validKey = tostring(config.Key or "changeme228")
	local title = tostring(config.KeySystemTitle or "Galaxy's Hub | Key System")
	local saveKey = config.SaveKey == true
	local fileName = tostring(config.FileName or "GalaxyHubKey.txt")

	local rememberedKey = nil
	if saveKey and type(isfile) == "function" and type(readfile) == "function" then
		local success, result = pcall(function()
			if isfile(fileName) then
				return readfile(fileName)
			end
		end)

		if success and type(result) == "string" then
			rememberedKey = result
		end
	end

	if rememberedKey == validKey then
		return true
	end

	local guiParent = getGuiParent()
	local oldGui = guiParent:FindFirstChild("GalaxysHubLibKeySystem")
	if oldGui then
		oldGui:Destroy()
	end

	local ScreenGui = create("ScreenGui", {
		Name = "GalaxysHubLibKeySystem",
		ResetOnSpawn = false,
		IgnoreGuiInset = false,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
		DisplayOrder = 1000000,
	}, guiParent)

	local MainFrame = create("Frame", {
		Name = "Main",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromOffset(390, 235),
		BackgroundColor3 = "@THEME::Background",
        BorderSizePixel = 0,
	}, ScreenGui)
	addCorner(MainFrame, 12)
	addStroke(MainFrame, Color3.fromRGB(100, 100, 120), 1, 0.25)

	local SizeConstraint = create("UISizeConstraint", {
		MinSize = Vector2.new(300, 220),
		MaxSize = Vector2.new(450, 280),
	}, MainFrame)

	local TopBar = create("Frame", {
		Name = "TopBar",
		Size = UDim2.new(1, 0, 0, 42),
		BackgroundColor3 = "@THEME::Surface",
		BorderSizePixel = 0,
	}, MainFrame)
	addCorner(TopBar, 12)
	makeDraggable(TopBar, MainFrame)

	create("TextLabel", {
		Name = "Title",
		Position = UDim2.fromOffset(14, 0),
		Size = UDim2.new(1, -58, 1, 0),
		BackgroundTransparency = 1,
		Text = title,
		TextColor3 = "@THEME::Text",
		TextSize = 15,
		Font = Enum.Font.GothamBold,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, TopBar)

	local CloseButton = create("TextButton", {
		Name = "Close",
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -9, 0.5, 0),
		Size = UDim2.fromOffset(30, 27),
		BackgroundColor3 = Color3.fromRGB(140, 45, 52),
		BorderSizePixel = 0,
		Text = "×",
		TextColor3 = Color3.new(1, 1, 1),
		TextSize = 18,
		Font = Enum.Font.GothamBold,
	}, TopBar)
	addCorner(CloseButton, 6)

	create("TextLabel", {
		Name = "Description",
		Position = UDim2.fromOffset(20, 55),
		Size = UDim2.new(1, -40, 0, 24),
		BackgroundTransparency = 1,
		Text = "Enter the key to continue.",
		TextColor3 = "@THEME::Muted",
		TextSize = 12,
		Font = Enum.Font.Gotham,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, MainFrame)

	local KeyBox = create("TextBox", {
		Name = "KeyBox",
		Position = UDim2.fromOffset(20, 85),
		Size = UDim2.new(1, -40, 0, 42),
		BackgroundColor3 = "@THEME::Raised",
		BorderSizePixel = 0,
		ClearTextOnFocus = false,
		PlaceholderText = "Enter key...",
		PlaceholderColor3 = "@THEME::Muted",
		Text = "",
		TextColor3 = "@THEME::Text",
		TextSize = 13,
		Font = Enum.Font.Gotham,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, MainFrame)
	addCorner(KeyBox, 7)
	addStroke(KeyBox, Color3.fromRGB(95, 95, 110), 1, 0.45)
	addPadding(KeyBox, 10, 10, 0, 0)

	local VerifyButton = create("TextButton", {
		Name = "Verify",
		Position = UDim2.fromOffset(20, 139),
		Size = UDim2.new(1, -40, 0, 38),
		BackgroundColor3 = "@THEME::Accent",
		BorderSizePixel = 0,
		Text = "VERIFY KEY",
		TextColor3 = Color3.new(1, 1, 1),
		TextSize = 13,
		Font = Enum.Font.GothamBold,
		AutoButtonColor = true,
	}, MainFrame)
	addCorner(VerifyButton, 7)

	local StatusLabel = create("TextLabel", {
		Name = "Status",
		Position = UDim2.fromOffset(20, 184),
		Size = UDim2.new(1, -40, 0, 28),
		BackgroundTransparency = 1,
		Text = "",
		TextColor3 = Color3.fromRGB(255, 105, 110),
		TextSize = 12,
		Font = Enum.Font.Gotham,
		TextWrapped = true,
	}, MainFrame)

	local verified = false
	local closed = false
	local checking = false

	local function verify()
		if checking or closed then
			return
		end

		checking = true
		local enteredKey = KeyBox.Text

		if enteredKey == validKey then
			StatusLabel.TextColor3 = Color3.fromRGB(75, 230, 130)
			StatusLabel.Text = "Correct key. Access granted!"

			if saveKey and type(writefile) == "function" then
				pcall(writefile, fileName, enteredKey)
			end

			verified = true
			task.wait(0.35)

			if ScreenGui.Parent then
				ScreenGui:Destroy()
			end
		else
			StatusLabel.TextColor3 = Color3.fromRGB(255, 105, 110)
			StatusLabel.Text = "Incorrect key."

			local originalPosition = MainFrame.Position
			for _, offset in ipairs({-8, 8, -6, 6, 0}) do
				MainFrame.Position = UDim2.new(
					originalPosition.X.Scale,
					originalPosition.X.Offset + offset,
					originalPosition.Y.Scale,
					originalPosition.Y.Offset
				)
				task.wait(0.035)
			end
			MainFrame.Position = originalPosition
		end

		checking = false
	end

	VerifyButton.MouseButton1Click:Connect(verify)

	KeyBox.FocusLost:Connect(function(enterPressed)
		if enterPressed then
			verify()
		end
	end)

	CloseButton.MouseButton1Click:Connect(function()
		closed = true
		if ScreenGui.Parent then
			ScreenGui:Destroy()
		end
	end)

	while not verified and not closed do
		task.wait()
	end

	return verified
end
